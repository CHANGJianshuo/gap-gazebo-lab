cmake_minimum_required(VERSION 3.16)
project(gap_gazebo_lab)
set(CMAKE_CXX_STANDARD 17)
# This small bridge is not the physics engine; avoid expensive O3 template optimization.
set(CMAKE_CXX_FLAGS_RELEASE "-O1 -DNDEBUG")
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(gap_world SHARED src/world.cpp)
target_link_libraries(gap_world gz-sim8::core gz-plugin2::register)
ament_target_dependencies(gap_world rclcpp std_msgs std_srvs)
