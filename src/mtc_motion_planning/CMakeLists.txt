cmake_minimum_required(VERSION 3.16)
project(mtc_motion_planning)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_planners_ompl REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(mtc_interfaces REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometric_shapes REQUIRED)
find_package(joint_trajectory_controller REQUIRED)
find_package(ruckig REQUIRED)
find_package(Eigen3 REQUIRED)
include_directories(include)
add_executable(trajectory_self_test test/test_trajectory.cpp)
add_executable(short_moves_test test/test_short_moves.cpp)
target_link_libraries(short_moves_test Eigen3::Eigen ruckig::ruckig)
target_link_libraries(trajectory_self_test Eigen3::Eigen ruckig::ruckig)
add_executable(planner src/planner.cpp)
add_executable(dynamics_probe test/dynamics_probe.cpp)
add_executable(controller_interpolation_test test/test_controller_interpolation.cpp)
ament_target_dependencies(controller_interpolation_test joint_trajectory_controller rclcpp)
target_link_libraries(controller_interpolation_test Eigen3::Eigen ruckig::ruckig)
ament_target_dependencies(dynamics_probe moveit_core rclcpp)
target_link_libraries(dynamics_probe Eigen3::Eigen ruckig::ruckig)
ament_target_dependencies(planner rclcpp rclcpp_action moveit_core moveit_ros_planning moveit_planners_ompl moveit_msgs mtc_interfaces tf2_msgs std_msgs sensor_msgs geometric_shapes)
target_link_libraries(planner Eigen3::Eigen ruckig::ruckig)
install(TARGETS planner DESTINATION lib/${PROJECT_NAME})
install(TARGETS trajectory_self_test short_moves_test dynamics_probe controller_interpolation_test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
